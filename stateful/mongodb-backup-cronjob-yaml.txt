apiVersion: batch/v1
kind: CronJob
metadata:
  name: mongodb-backup
  namespace: multi-tier
spec:
  schedule: "0 2 * * *"  # Runs at 2 AM every day
  successfulJobsHistoryLimit: 3
  failedJobsHistoryLimit: 1
  concurrencyPolicy: Forbid
  jobTemplate:
    spec:
      template:
        spec:
          containers:
          - name: mongodb-backup
            image: mongo:5.0
            command:
            - /bin/bash
            - -c
            - |
              TIMESTAMP=$(date +%Y%m%d_%H%M%S)
              BACKUP_NAME="mongodb_backup_$TIMESTAMP"
              echo "Starting backup: $BACKUP_NAME"
              
              # Create backup using mongodump
              mongodump --host=mongodb.multi-tier.svc.cluster.local:27017 \
                        --username=root \
                        --password=$MONGODB_ROOT_PASSWORD \
                        --authenticationDatabase=admin \
                        --db=app-database \
                        --out=/backup/$BACKUP_NAME
              
              echo "Backup completed: $BACKUP_NAME"
              
              # Keep only the last 7 backups
              ls -tp /backup/ | grep -v '/$' | tail -n +8 | xargs -I {} rm -- /backup/{}
            env:
            - name: MONGODB_ROOT_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: mongodb-secret
                  key: root-password
            volumeMounts:
            - name: backup-storage
              mountPath: /backup
          restartPolicy: OnFailure
          volumes:
          - name: backup-storage
            persistentVolumeClaim:
              claimName: mongodb-backup-pvc
