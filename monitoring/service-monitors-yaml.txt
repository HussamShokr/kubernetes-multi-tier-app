apiVersion: monitoring.coreos.com/v1
kind: ServiceMonitor
metadata:
  name: frontend-monitor
  namespace: monitoring
  labels:
    release: prometheus
spec:
  selector:
    matchLabels:
      app: frontend
      tier: frontend
  namespaceSelector:
    matchNames:
      - multi-tier
  endpoints:
  - port: http
    interval: 15s
    path: /metrics
---
